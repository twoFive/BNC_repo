' ============================================================================
'  frm_Log
'  Historia wszystkich zgłoszeń (pending + sent) + eksport DataCache do xlsx.
' ============================================================================

Option Explicit

' ----- Lifecycle -----------------------------------------------------------

' LoadRecords w Activate (nie Initialize) bo Activate fires przy KAŻDYM Show.
' Bez tego widok byłby stały gdy user usunie pending w frm_Main i wróci
' do frm_Log (patrz ADR-007).
Private Sub UserForm_Activate()
    LoadRecords
End Sub

' ----- Buttons -------------------------------------------------------------

Private Sub btn_Export_Click()
    Dim suggested As String
    suggested = mod_Export.GetSuggestedExportFileName()

    Dim targetPath As String
    targetPath = ShowSaveDialog(suggested)
    If Len(targetPath) = 0 Then Exit Sub  ' user anulował

    Application.Cursor = xlWait
    Dim ok As Boolean
    ok = mod_Export.ExportDataCache(targetPath)
    Application.Cursor = xlDefault

    If ok Then
        MsgBox "Wyeksportowano do:" & vbCrLf & targetPath, _
               vbInformation, "Eksport"
    Else
        MsgBox "Błąd eksportu. Sprawdź Immediate Window (Ctrl+G) dla szczegółów.", _
               vbExclamation, "Błąd eksportu"
    End If
End Sub

' Hub-and-spoke: zamknięcie = return do sh_LandingPage.
Private Sub btn_Back_Click()
    Me.Hide
End Sub

' ----- Helpers -------------------------------------------------------------

Private Sub LoadRecords()
    Dim allRecords As Collection
    Set allRecords = mod_DataCacheSync.GetAllRecords()

    Dim pendingCount As Long, sentCount As Long
    Dim record As Object
    For Each record In allRecords
        If CStr(record("Status")) = "pending" Then
            pendingCount = pendingCount + 1
        Else
            sentCount = sentCount + 1
        End If
    Next record

    lbl_Stats.Caption = "Wszystkich: " & allRecords.Count & _
                       "   |   Pending: " & pendingCount & _
                       "   |   Sent: " & sentCount

    lst_AllRecords.Clear
    If allRecords.Count = 0 Then Exit Sub

    ' 7 kolumn: ID | FK | Nazwa | CNA | Status | Wysłany | Data wysłania
    ' Najnowsze na górze (odwracamy kolejność).
    Dim n As Long
    n = allRecords.Count
    Dim arr() As Variant
    ReDim arr(0 To n - 1, 0 To 6)

    Dim i As Long, idx As Long
    i = 0
    For Each record In allRecords
        idx = n - 1 - i  ' odwrotna kolejność - najnowsze na górze
        arr(idx, 0) = record("ReportID")
        arr(idx, 1) = record("KlientFK")
        arr(idx, 2) = record("NazwaKlienta")
        arr(idx, 3) = record("CNA_HandlowcaID")
        arr(idx, 4) = record("Status")
        arr(idx, 5) = FormatRecipient(CStr(record("EmailRecipient")))
        arr(idx, 6) = FormatTimestamp(record("BatchSentTimestamp"))
        i = i + 1
    Next record

    lst_AllRecords.List = arr
End Sub

' Recipient pusty (pending) -> "-".
Private Function FormatRecipient(s As String) As String
    If Len(Trim$(s)) = 0 Then
        FormatRecipient = "-"
    Else
        FormatRecipient = s
    End If
End Function

' Timestamp pusty (pending) -> "-", inaczej yyyy-mm-dd.
Private Function FormatTimestamp(v As Variant) As String
    If IsEmpty(v) Or v = "" Then
        FormatTimestamp = "-"
    ElseIf IsDate(v) Then
        FormatTimestamp = Format(v, "yyyy-mm-dd")
    Else
        FormatTimestamp = CStr(v)
    End If
End Function

Private Function ShowSaveDialog(defaultName As String) As String
    Dim result As Variant
    result = Application.GetSaveAsFilename( _
        InitialFileName:=defaultName, _
        FileFilter:="Excel Files (*.xlsx), *.xlsx", _
        Title:="Zapisz eksport jako")

    If TypeName(result) = "Boolean" Then
        ShowSaveDialog = ""  ' user anulował
    Else
        ShowSaveDialog = CStr(result)
    End If
End Function
