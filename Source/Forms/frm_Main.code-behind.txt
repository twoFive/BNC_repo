Private Sub btn_AddToList_Click()
    Dim reportData As Object
    Set reportData = CreateObject("Scripting.Dictionary")
    reportData("KlientFK") = Trim$(txt_KlientFK.Text)
    reportData("NazwaKlienta") = Trim$(txt_NazwaKlienta.Text)
    reportData("MiesiacObrotu") = Trim$(txt_MiesiacObrotu.Text)

    Dim errMsg As String
    errMsg = mod_Validation.ValidateReportData(reportData)
    If Len(errMsg) > 0 Then
        MsgBox errMsg, vbExclamation, "B³¹d walidacji"
        Exit Sub
    End If

    Dim newID As Long
    newID = mod_DataCacheSync.AppendRecord(reportData)
    mod_Utils.LogInfo "Dodano zgloszenie ID=" & newID

    ClearFormFields
    RefreshPendingList
End Sub

Private Sub btn_SendBatch_Click()
    Dim pending As Collection
    Set pending = mod_DataCacheSync.GetPendingRecords()

    If pending.Count = 0 Then
        MsgBox "Brak zg³oszeñ do wys³ania.", vbInformation, "BNC_Sender"
        Exit Sub
    End If

    Dim msg As String
    If mod_UsersRegistrySync.IsUserManager() Then
        msg = "Wys³aæ " & pending.Count & " zg³oszeñ wprost do BNC?"
    Else
        msg = "Wys³aæ " & pending.Count & " zg³oszeñ do kierownika " & _
              "(" & CStr(mod_UsersRegistrySync.GetCurrentUserField("EmailKierownika")) & ")?"
    End If
    If MsgBox(msg, vbYesNo + vbQuestion, "Potwierdzenie wysy³ki") <> vbYes Then Exit Sub

    ' Disable przyciski podczas wysy³ki - prevent double-click.
    DisableActionButtons True
    Application.Cursor = xlWait

    Dim success As Boolean
    success = mod_MailSender.SendBatch()

    Application.Cursor = xlDefault
    DisableActionButtons False

    If success Then
        MsgBox "Batch wys³any pomyœlnie.", vbInformation, "BNC_Sender"
        RefreshPendingList
    Else
        MsgBox "B³¹d wysy³ki. SprawdŸ Immediate Window (Ctrl+G) dla szczegó³ów.", _
               vbExclamation, "B³¹d wysy³ki"
    End If
End Sub

' Hard delete pending recordu (ADR-006). Potwierdzenie z ID + nazw¹ klienta
' ¿eby user mia³ szansê zatrzymaæ siê przed pomy³k¹.
Private Sub btn_DeleteSelected_Click()
    If lst_PendingBatch.ListIndex < 0 Then
        MsgBox "Najpierw zaznacz zg³oszenie do usuniêcia z listy.", _
               vbInformation, "Brak zaznaczenia"
        Exit Sub
    End If

    Dim idx As Long
    idx = lst_PendingBatch.ListIndex

    Dim reportID As Long
    Dim nazwa As String
    reportID = CLng(lst_PendingBatch.List(idx, 0))
    nazwa = CStr(lst_PendingBatch.List(idx, 2))

    Dim msg As String
    msg = "Potwierdzenie usuniêcia zg³oszenia BNC" & vbCrLf & vbCrLf & _
          "ID: " & reportID & vbCrLf & _
          "Klient: " & nazwa & vbCrLf & vbCrLf & _
          "Operacja nieodwracalna. Kontynuowaæ?"

    Dim answer As VbMsgBoxResult
    answer = MsgBox(msg, vbYesNo + vbQuestion + vbDefaultButton2, "Potwierdzenie")
    If answer <> vbYes Then Exit Sub

    If mod_DataCacheSync.DeleteRecord(reportID) Then
        mod_Utils.LogInfo "Usunieto zgloszenie ID=" & reportID
        RefreshPendingList
    Else
        MsgBox "Nie uda³o siê usun¹æ zg³oszenia. SprawdŸ Immediate Window (Ctrl+G).", _
               vbExclamation, "B³¹d usuwania"
    End If
End Sub

' ----- Helpers -------------------------------------------------------------

Private Sub ClearFormFields()
    txt_KlientFK.Text = ""
    txt_NazwaKlienta.Text = ""
    ' MiesiacObrotu zostaje (user mo¿e dodaæ kilka zg³oszeñ na ten sam miesi¹c).
    On Error Resume Next
    txt_KlientFK.SetFocus
    On Error GoTo 0
End Sub

' Odœwie¿enie listy pending. Buduje 2D array i przypisuje do ListBox.List
' jednym ruchem - ~10x szybsze ni¿ AddItem w pêtli.
'
' Newest first - iterujemy od koñca do pocz¹tku, bo GetPendingRecords zwraca
' rekordy w kolejnoœci arkusza (najstarsze pierwsze).
Private Sub RefreshPendingList()
    Dim pending As Collection
    Set pending = mod_DataCacheSync.GetPendingRecords()

    lbl_BatchCount.Caption = "Lista zg³oszeñ do wys³ania (" & pending.Count & ")"
    btn_DeleteSelected.Enabled = (pending.Count > 0)

    lst_PendingBatch.Clear
    If pending.Count = 0 Then Exit Sub

    ' 5 kolumn: ID, KlientFK, Nazwa, MiesiacObrotu, CNA
    Dim arr() As Variant
    ReDim arr(0 To pending.Count - 1, 0 To 4)

    Dim i As Long
    Dim record As Object
    i = pending.Count - 1
    For Each record In pending
        arr(i, 0) = record("ReportID")
        arr(i, 1) = record("KlientFK")
        arr(i, 2) = record("NazwaKlienta")
        arr(i, 3) = record("MiesiacObrotu")
        arr(i, 4) = record("CNA_HandlowcaID")
        i = i - 1
    Next record

    lst_PendingBatch.List = arr
End Sub

' Disable Add/Send/Delete podczas wysy³ki. Po Disable=False stan
' btn_DeleteSelected dostaje z RefreshPendingList (pending.Count > 0).
Private Sub DisableActionButtons(disabled As Boolean)
    btn_AddToList.Enabled = Not disabled
    btn_SendBatch.Enabled = Not disabled
    btn_Clear.Enabled = Not disabled
    btn_ShowLog.Enabled = Not disabled
    btn_DeleteSelected.Enabled = Not disabled
End Sub
