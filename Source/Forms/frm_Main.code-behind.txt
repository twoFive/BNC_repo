' ============================================================================
'  frm_Main
'  Główny formularz: dodawanie zgłoszeń + wysyłka batcha.
'
'  ListBox refresh przez wlaściwość List (2D array) - ~10x szybsze niż
'  AddItem + List(i, j) w pętli, bez migotania.
' ============================================================================

Option Explicit

' ----- Lifecycle -----------------------------------------------------------

' Initialize - static setup (raz na instancję):
' identity caption, role detection, default miesiąc obrotu.
Private Sub UserForm_Initialize()
    lbl_UserInfo.Caption = "Zalogowany: " & _
        CStr(mod_UsersRegistrySync.GetCurrentUserField("Imie")) & " " & _
        CStr(mod_UsersRegistrySync.GetCurrentUserField("Nazwisko")) & _
        " (CNA: " & CStr(mod_UsersRegistrySync.GetCurrentUserField("CNA_HandlowcaID")) & _
        ", oddział: " & CStr(mod_UsersRegistrySync.GetCurrentUserField("NrOddzialu")) & ")"

    If mod_UsersRegistrySync.IsUserManager() Then
        lbl_RoleInfo.Caption = "Tryb: KIEROWNIK (wnioski wysyłane wprost do BNC)"
    Else
        lbl_RoleInfo.Caption = "Tryb: HANDLOWIEC (wnioski wysyłane do kierownika)"
    End If

    txt_MiesiacObrotu.Text = Format(mod_Utils.GetCurrentMonthYear(), "yyyy-mm")
    ' Fake header labels (lbl_HdrID/FK/Nazwa/MiesiacObrotu/CNA) mają Captions
    ' ustawione statycznie w Designerze - zero kodu runtime.
End Sub

' Activate - dynamic refresh (przy każdym Show, także po powrocie z frm_Log).
Private Sub UserForm_Activate()
    RefreshPendingList
End Sub

' ----- Buttons -------------------------------------------------------------

Private Sub btn_AddToList_Click()
    Dim reportData As Object
    Set reportData = CreateObject("Scripting.Dictionary")
    reportData("KlientFK") = Trim$(txt_KlientFK.Text)
    reportData("NazwaKlienta") = Trim$(txt_NazwaKlienta.Text)
    reportData("MiesiacObrotu") = Trim$(txt_MiesiacObrotu.Text)

    Dim errMsg As String
    errMsg = mod_Validation.ValidateReportData(reportData)
    If Len(errMsg) > 0 Then
        MsgBox errMsg, vbExclamation, "Błąd walidacji"
        Exit Sub
    End If

    Dim newID As Long
    newID = mod_DataCacheSync.AppendRecord(reportData)
    mod_Utils.LogInfo "Dodano zgloszenie ID=" & newID

    ClearFormFields
    RefreshPendingList
End Sub

Private Sub btn_Clear_Click()
    ClearFormFields
End Sub

Private Sub btn_SendBatch_Click()
    Dim pending As Collection
    Set pending = mod_DataCacheSync.GetPendingRecords()

    If pending.Count = 0 Then
        MsgBox "Brak zgłoszeń do wysłania.", vbInformation, "BNC_Sender"
        Exit Sub
    End If

    Dim msg As String
    If mod_UsersRegistrySync.IsUserManager() Then
        msg = "Wysłać " & pending.Count & " zgłoszeń wprost do BNC?"
    Else
        msg = "Wysłać " & pending.Count & " zgłoszeń do kierownika " & _
              "(" & CStr(mod_UsersRegistrySync.GetCurrentUserField("EmailKierownika")) & ")?"
    End If
    If MsgBox(msg, vbYesNo + vbQuestion, "Potwierdzenie wysyłki") <> vbYes Then Exit Sub

    ' Disable przyciski podczas wysyłki - prevent double-click.
    DisableActionButtons True
    Application.Cursor = xlWait

    Dim success As Boolean
    success = mod_MailSender.SendBatch()

    Application.Cursor = xlDefault
    DisableActionButtons False

    If success Then
        MsgBox "Batch wysłany pomyślnie.", vbInformation, "BNC_Sender"
        RefreshPendingList
    Else
        MsgBox "Błąd wysyłki. Sprawdź Immediate Window (Ctrl+G) dla szczegółów.", _
               vbExclamation, "Błąd wysyłki"
    End If
End Sub

Private Sub btn_ShowLog_Click()
    If mod_Utils.IsFormOpen("frm_Log") Then Exit Sub
    Me.Hide
    frm_Log.Show vbModal
End Sub

' Hard delete pending recordu (ADR-006). Potwierdzenie z ID + nazwą klienta
' żeby user miał szansę zatrzymać się przed pomyłką.
Private Sub btn_DeleteSelected_Click()
    If lst_PendingBatch.ListIndex < 0 Then
        MsgBox "Najpierw zaznacz zgłoszenie do usunięcia z listy.", _
               vbInformation, "Brak zaznaczenia"
        Exit Sub
    End If

    Dim idx As Long
    idx = lst_PendingBatch.ListIndex

    Dim reportID As Long
    Dim nazwa As String
    reportID = CLng(lst_PendingBatch.List(idx, 0))
    nazwa = CStr(lst_PendingBatch.List(idx, 2))

    Dim msg As String
    msg = "Potwierdzenie usunięcia zgłoszenia BNC" & vbCrLf & vbCrLf & _
          "ID: " & reportID & vbCrLf & _
          "Klient: " & nazwa & vbCrLf & vbCrLf & _
          "Operacja nieodwracalna. Kontynuować?"

    Dim answer As VbMsgBoxResult
    answer = MsgBox(msg, vbYesNo + vbQuestion + vbDefaultButton2, "Potwierdzenie")
    If answer <> vbYes Then Exit Sub

    If mod_DataCacheSync.DeleteRecord(reportID) Then
        mod_Utils.LogInfo "Usunieto zgloszenie ID=" & reportID
        RefreshPendingList
    Else
        MsgBox "Nie udało się usunąć zgłoszenia. Sprawdź Immediate Window (Ctrl+G).", _
               vbExclamation, "Błąd usuwania"
    End If
End Sub

' ----- Helpers -------------------------------------------------------------

Private Sub ClearFormFields()
    txt_KlientFK.Text = ""
    txt_NazwaKlienta.Text = ""
    ' MiesiacObrotu zostaje (user może dodać kilka zgłoszeń na ten sam miesiąc).
    On Error Resume Next
    txt_KlientFK.SetFocus
    On Error GoTo 0
End Sub

' Odświeżenie listy pending. Buduje 2D array i przypisuje do ListBox.List
' jednym ruchem - ~10x szybsze niż AddItem w pętli.
'
' Newest first - iterujemy od końca do początku, bo GetPendingRecords zwraca
' rekordy w kolejności arkusza (najstarsze pierwsze).
Private Sub RefreshPendingList()
    Dim pending As Collection
    Set pending = mod_DataCacheSync.GetPendingRecords()

    lbl_BatchCount.Caption = "Lista zgłoszeń do wysłania (" & pending.Count & ")"
    btn_DeleteSelected.Enabled = (pending.Count > 0)

    lst_PendingBatch.Clear
    If pending.Count = 0 Then Exit Sub

    ' 5 kolumn: ID, KlientFK, Nazwa, MiesiacObrotu, CNA
    Dim arr() As Variant
    ReDim arr(0 To pending.Count - 1, 0 To 4)

    Dim i As Long
    Dim record As Object
    i = pending.Count - 1
    For Each record In pending
        arr(i, 0) = record("ReportID")
        arr(i, 1) = record("KlientFK")
        arr(i, 2) = record("NazwaKlienta")
        arr(i, 3) = record("MiesiacObrotu")
        arr(i, 4) = record("CNA_HandlowcaID")
        i = i - 1
    Next record

    lst_PendingBatch.List = arr
End Sub

' Disable Add/Send/Delete podczas wysyłki. Po Disable=False stan
' btn_DeleteSelected dostaje z RefreshPendingList (pending.Count > 0).
Private Sub DisableActionButtons(disabled As Boolean)
    btn_AddToList.Enabled = Not disabled
    btn_SendBatch.Enabled = Not disabled
    btn_Clear.Enabled = Not disabled
    btn_ShowLog.Enabled = Not disabled
    btn_DeleteSelected.Enabled = Not disabled
End Sub
