Option Explicit

Private Const TOTAL_PAGES As Long = 5
Private m_currentPage As Long  ' 1..TOTAL_PAGES

' Screenshot area toggle - strony 3, 4 maja image, 1/2/5 text-only.
' TODO: kod screenshot toggle wylaczony do momentu dodania kontrolek
' img_ScreenshotMain i img_ScreenshotLog w Designerze frm_Tutorial.
' Uncomment blok "Screenshot area" w RenderPage + HideAllScreenshots po dodaniu.
Private Const BODY_HEIGHT_TEXT_ONLY As Long = 430

' ----- Lifecycle -----------------------------------------------------------

Private Sub UserForm_Initialize()
    m_currentPage = 1
    RenderPage
End Sub

' ----- Buttons -------------------------------------------------------------

Private Sub btn_Next_Click()
    If m_currentPage < TOTAL_PAGES Then
        m_currentPage = m_currentPage + 1
        RenderPage
    Else
        Me.Hide  ' ostatnia strona: Next == Zakoncz
    End If
End Sub

Private Sub btn_Prev_Click()
    If m_currentPage > 1 Then
        m_currentPage = m_currentPage - 1
        RenderPage
    End If
End Sub

Private Sub btn_Skip_Click()
    Me.Hide
End Sub

' ----- Rendering -----------------------------------------------------------

' Dispatcher wolany po kazdej zmianie strony.
Private Sub RenderPage()
    lbl_PageIndicator.Caption = "Strona " & m_currentPage & " z " & TOTAL_PAGES
    btn_Prev.Enabled = (m_currentPage > 1)

    If m_currentPage = TOTAL_PAGES Then
        btn_Next.Caption = "Zakoncz"
    Else
        btn_Next.Caption = "Dalej ->"
    End If

    ' Screenshot area - stale text-only do momentu dodania Image controls
    ' (img_ScreenshotMain, img_ScreenshotLog) w Designerze frm_Tutorial.
    txt_PageBody.Height = BODY_HEIGHT_TEXT_ONLY

    ' TODO: po dodaniu Image controls, uncomment blok ponizej i usun
    ' powyzsze przypisanie BODY_HEIGHT_TEXT_ONLY:
    '
    ' Dim hasImage As Boolean
    ' hasImage = (m_currentPage = 3 Or m_currentPage = 4)
    ' HideAllScreenshots
    ' If hasImage Then
    '     txt_PageBody.Height = BODY_HEIGHT_WITH_IMAGE
    '     Select Case m_currentPage
    '         Case 3: img_ScreenshotMain.Visible = True
    '         Case 4: img_ScreenshotLog.Visible = True
    '     End Select
    ' Else
    '     txt_PageBody.Height = BODY_HEIGHT_TEXT_ONLY
    ' End If

    Select Case m_currentPage
        Case 1: RenderPage1_Welcome
        Case 2: RenderPage2_Setup
        Case 3: RenderPage3_DailyWork
        Case 4: RenderPage4_HistoryExport
        Case 5: RenderPage5_Help
    End Select
End Sub
