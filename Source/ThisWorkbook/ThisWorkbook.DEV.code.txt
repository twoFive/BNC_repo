Option Explicit

Private Sub Workbook_Open()
    On Error GoTo ErrorHandler

    ' 1. Ensure app state sheet (ADR-009 session marker _CurrentUserID)
    mod_AppStateSync.EnsureAppStateSheet

    ' 2. Auto-recreate cache xlsx (no-op gdy CacheFolderPath nieustawiony)
    mod_DataCacheSync.EnsureCacheFileExists              ' BNC_DataCache.xlsx
    mod_UsersRegistrySync.EnsureRegistryCacheFileExists  ' BNC_UsersRegistry.xlsx

    ' 3. Routing multi-user
    If mod_UsersRegistrySync.GetUsersCount() = 0 Then
        mod_Utils.LogInfo "Workbook_Open [DEV]: Registry pusty, pokazuje frm_Setup"
        frm_Setup.Show vbModal
    Else
        mod_Utils.LogInfo "Workbook_Open [DEV]: " & mod_UsersRegistrySync.GetUsersCount() & _
                         " uzytkownikow, pokazuje frm_UserPicker"
        frm_UserPicker.Show vbModal
    End If

    ' 4. Landing na hub. Worksheet_Activate triggeruje RefreshDashboard.
    sh_LandingPage.Activate
    Exit Sub

ErrorHandler:
    mod_Utils.LogError "ThisWorkbook.Workbook_Open [DEV]", Err.Number, Err.Description
    MsgBox "Blad podczas startu aplikacji: " & Err.Description, _
           vbExclamation, "BNC_Sender [DEV]"
End Sub
