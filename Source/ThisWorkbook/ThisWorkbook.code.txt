' ============================================================================
'  ThisWorkbook
'  Wklej do klasy ThisWorkbook w VBE (specjalny modul, nie import).
'
'  Routing (M3.3 multi-user + M6 hub pattern):
'  - GetUsersCount = 0 -> frm_Setup (pierwszy raz)
'  - GetUsersCount >= 1 -> frm_UserPicker
'  - Po zamknieciu picker/setup -> sh_LandingPage (hub)
'
'  App-like UI: DisplayFullScreen = True przy Open, restore przy Close.
' ============================================================================

Option Explicit

Private Sub Workbook_Open()
    On Error GoTo ErrorHandler

    ' 1. Ensure app state sheet (ADR-009 session marker _CurrentUserID)
    mod_AppStateSync.EnsureAppStateSheet

    ' 2. Auto-recreate cache xlsx (no-op gdy CacheFolderPath nieustawiony)
    mod_DataCacheSync.EnsureCacheFileExists              ' BNC_DataCache.xlsx
    mod_UsersRegistrySync.EnsureRegistryCacheFileExists  ' BNC_UsersRegistry.xlsx

    ' 3. App-like UI: ukryj Excel chrome (ribbon, formula bar, headings, tabs)
    Application.DisplayFullScreen = True
    Application.DisplayFormulaBar = False
    ActiveWindow.DisplayHeadings = False
    ActiveWindow.DisplayGridlines = False
    ActiveWindow.DisplayWorkbookTabs = False
    ' Ribbon calkowicie hidden (nie tylko zminimalizowany) - Excel 4 macro
    ' bo brak natywnej Application property. Restore w BeforeClose.
    Application.ExecuteExcel4Macro "SHOW.TOOLBAR(""Ribbon"",False)"

    ' 4. Routing multi-user
    If mod_UsersRegistrySync.GetUsersCount() = 0 Then
        mod_Utils.LogInfo "Workbook_Open: Registry pusty, pokazuje frm_Setup"
        frm_Setup.Show vbModal
    Else
        mod_Utils.LogInfo "Workbook_Open: " & mod_UsersRegistrySync.GetUsersCount() & _
                         " uzytkownikow, pokazuje frm_UserPicker"
        frm_UserPicker.Show vbModal
    End If

    ' 5. Landing na hub. Worksheet_Activate triggeruje RefreshDashboard.
    sh_LandingPage.Activate

    ' 6. Sheet lock: user nie moze edytowac komorek ani ich zaznaczac.
    '    UserInterfaceOnly:=True pozwala VBA (RefreshDashboard) pisac do
    '    komorek mimo protekcji. ScrollArea ogranicza scroll do landing area.
    LockLandingPage
    Exit Sub

ErrorHandler:
    mod_Utils.LogError "ThisWorkbook.Workbook_Open", Err.Number, Err.Description
    MsgBox "Blad podczas startu aplikacji: " & Err.Description, _
           vbExclamation, "BNC_Sender"
End Sub

' Restore Excel chrome przy zamknieciu zeby inne pliki Excel otwarte
' rownolegle nie dziedziczyly DisplayFullScreen + hidden ribbon.
Private Sub Workbook_BeforeClose(Cancel As Boolean)
    On Error Resume Next
    Application.DisplayFullScreen = False
    Application.DisplayFormulaBar = True
    Application.ExecuteExcel4Macro "SHOW.TOOLBAR(""Ribbon"",True)"

    ' Odkomentuj zeby wylaczyc pytanie "Zapisac zmiany?" przy zamykaniu xlsm.
    ' Bezpieczne: aplikacja robi auto-save po kazdej mutacji (SetAppValue,
    ' AppendRecord, MarkAsSent, AddNewUser, SetCurrentUserField, itd.),
    ' wiec Saved=True nie gubi danych - tylko suppresses save prompt.
    ' ThisWorkbook.Saved = True
End Sub

' Cell protection dla landing page. Wywolywane raz w Workbook_Open po
' aktywacji sheeta. UserInterfaceOnly=True: VBA pisze do komorek, user nie.
Private Sub LockLandingPage()
    On Error Resume Next
    With sh_LandingPage
        .Unprotect  ' na wypadek istniejacej protekcji (bez hasla)
        .Cells.Locked = True
        .EnableSelection = xlNoSelection
        .ScrollArea = "A1:H23"
        .Protect Password:="", UserInterfaceOnly:=True, _
                 DrawingObjects:=True, Contents:=True, Scenarios:=True
    End With
End Sub
