Option Explicit

' ----- Lifecycle -----------------------------------------------------------

' Wolane przy KAZDEJ aktywacji sheeta (po zamknieciu formularza, po Workbook_Open,
' po klikni w tab). Refresh user info + stats zeby zawsze aktualne.
Private Sub Worksheet_Activate()
    RefreshDashboard
End Sub

' ----- Rendering -----------------------------------------------------------

' Aktualizuje dynamic cells: user info (B3, B4, B5) + statystyki (B20, E20).
' Wolane z Worksheet_Activate. Public zeby inne moduly mogly wywolac po
' zmianie stanu (np. mod_LandingPage handlery gdyby chcialy force refresh).
Public Sub RefreshDashboard()
    On Error GoTo Cleanup

    ' User info z Registry (przez current user API)
    Dim imie As String, nazwisko As String
    Dim cna As String, oddzial As String

    imie = CStr(mod_UsersRegistrySync.GetCurrentUserField("Imie"))
    nazwisko = CStr(mod_UsersRegistrySync.GetCurrentUserField("Nazwisko"))
    cna = CStr(mod_UsersRegistrySync.GetCurrentUserField("CNA_HandlowcaID"))
    oddzial = CStr(mod_UsersRegistrySync.GetCurrentUserField("NrOddzialu"))

    If Len(imie) = 0 Then
        ' Brak aktywnego usera (edge case - _CurrentUserID puste)
        Range("B3").Value = "(brak aktywnego uzytkownika)"
        Range("B4").Value = ""
        Range("B5").Value = "Wybierz uzytkownika przez 'Przelacz usera'"
    Else
        Range("B3").Value = imie & " " & nazwisko
        Range("B4").Value = cna & " · Oddzial: " & oddzial

        If mod_UsersRegistrySync.IsUserManager() Then
            Range("B5").Value = "KIEROWNIK (mail leci wprost do BNC)"
        Else
            Range("B5").Value = "HANDLOWIEC (mail leci do kierownika: " & _
                CStr(mod_UsersRegistrySync.GetCurrentUserField("EmailKierownika")) & ")"
        End If
    End If

    ' Statystyki z DataCache
    Dim pending As Collection, allRecs As Collection
    Set pending = mod_DataCacheSync.GetPendingRecords()
    Set allRecs = mod_DataCacheSync.GetAllRecords()

    Range("B20").Value = pending.Count
    Range("E20").Value = allRecs.Count - pending.Count  ' sent = all - pending

    Exit Sub

Cleanup:
    ' Best-effort refresh - jesli cos padnie (np. registry pusty), po prostu
    ' zostawiamy stare wartosci i logujemy blad. Nie chcemy zeby dashboard
    ' padnal przy niekompletnym setupie.
    mod_Utils.LogError "sh_LandingPage.RefreshDashboard", Err.Number, Err.Description
End Sub
